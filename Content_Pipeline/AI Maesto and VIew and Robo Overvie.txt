                <p id="modal-desc" class="text-slate-300 leading-relaxed mb-6">
                    Description goes here.
                </p>
                
                <div class="grid grid-cols-1 sm:grid-cols-2 gap-4 mb-8">
                    <div class="bg-dark-900/50 p-4 rounded-xl border border-white/5">
                        <span class="text-xs text-slate-500 uppercase font-bold block mb-1">Contact</span>
                        <span id="modal-contact" class="text-white font-medium">contact@example.com</span>
                    </div>
                    <div class="bg-dark-900/50 p-4 rounded-xl border border-white/5">
                        <span class="text-xs text-slate-500 uppercase font-bold block mb-1">Specialty</span>
                        <span id="modal-specialty" class="text-white font-medium">Integration</span>
                    </div>
                </div>

                <div class="flex gap-4">
                    <a id="modal-link" href="#" target="_blank" class="flex-1 bg-brand-600 hover:bg-brand-500 text-white text-center py-3 rounded-xl font-semibold transition-colors">
                        Visit Website <i class="fa-solid fa-external-link-alt ml-2 text-xs"></i>

                        Close

    <!-- JAVASCRIPT LOGIC -->
    <script>
        // --- DATA MOCKUP ---
        // Generating 25 mock companies for each category to demonstrate pagination
        const generateCompanies = (category, iconClass, colorClass) => {
            const companies = [];
            const specialties = ['Integration', 'Consulting', 'Security', 'Development', 'Analytics'];
            
            for (let i = 1; i <= 25; i++) {
                companies.push({
                    id: `${category}-${i}`,
                    name: `${category.charAt(0).toUpperCase() + category.slice(1)} Partner ${i}`,
                    category: category,
                    description: `Leading provider of ${category} solutions. Specializes in enterprise-grade deployments, offering robust support and scalable architecture for autonomous agent ecosystems.`,
                    contact: `partner${i}@${category}.com`,
                    specialty: specialties[Math.floor(Math.random() * specialties.length)],
                    website: `https://example.com/${category}/${i}`,
                    icon: iconClass,
                    color: colorClass
                });
            }
            return companies;
        };

        const allCompanies = [
            ...generateCompanies('maestro', 'fa-brain', 'text-indigo-400'),
            ...generateCompanies('agentsview', 'fa-shield-halved', 'text-emerald-400'),
            ...generateCompanies('roborev', 'fa-vial-circle-check', 'text-rose-400')
        ];

        // --- STATE MANAGEMENT ---
        const state = {
            maestro: { page: 1, itemsPerPage: 10 },
            agentsview: { page: 1, itemsPerPage: 10 },
            roborev: { page: 1, itemsPerPage: 10 }
        };

        // --- VIEW LOGIC ---
        function showView(viewName) {
            // Hide all views
            document.querySelectorAll('.view-section').forEach(el => {
                el.classList.add('hidden');
            });
            
            // Show selected view
            const target = document.getElementById(`view-${viewName}`);
            if (target) {
                target.classList.remove('hidden');
                window.scrollTo(0, 0);
                
                // Initialize charts or lists if needed
                if (viewName === 'home' && !window.homeChartInstance) {
                    initHomeChart();
                } else if (viewName === 'maestro') {
                    renderCategory('maestro');
                } else if (viewName === 'agentsview') {
                    renderCategory('agentsview');
                } else if (viewName === 'roborev') {
                    renderCategory('roborev');
                }
            }

            // Close mobile menu if open
            document.getElementById('mobile-menu').classList.add('hidden');
        }

        function toggleMobileMenu() {
            const menu = document.getElementById('mobile-menu');
            menu.classList.toggle('hidden');
        }

        // --- RENDER LOGIC ---
        function renderCategory(category) {
            const container = document.getElementById(`${category}-container`);
            const companies = allCompanies.filter(c => c.category === category);
            const currentState = state[category];
            
            // Pagination Logic
            const totalPages = Math.ceil(companies.length / currentState.itemsPerPage);
            const start = (currentState.page - 1) * currentState.itemsPerPage;
            const end = start + currentState.itemsPerPage;
            const pageItems = companies.slice(start, end);

            // Build HTML
            let html = `
                <div class="flex items-center justify-between mb-6">
                    <h3 class="text-xl text-white font-display">Partner Listings <span class="text-slate-500 text-sm font-sans ml-2">(${companies.length} total)</span></h3>
                    <div class="flex gap-2">
                        <button onclick="changePage('${category}', -1)" ${currentState.page === 1 ? 'disabled' : ''} class="w-8 h-8 rounded-lg bg-dark-800 border border-white/10 flex items-center justify-center hover:bg-brand-600 hover:text-white disabled:opacity-50 disabled:hover:bg-dark-800 disabled:hover:text-slate-400 transition-colors">
                            <i class="fa-solid fa-chevron-left text-xs"></i>
                        </button>
                        <span class="px-3 py-1 bg-dark-800 rounded-lg text-sm text-white border border-white/10">Page ${currentState.page} / ${totalPages}</span>
                        <button onclick="changePage('${category}', 1)" ${currentState.page === totalPages ? 'disabled' : ''} class="w-8 h-8 rounded-lg bg-dark-800 border border-white/10 flex items-center justify-center hover:bg-brand-600 hover:text-white disabled:opacity-50 disabled:hover:bg-dark-800 disabled:hover:text-slate-400 transition-colors">
                            <i class="fa-solid fa-chevron-right text-xs"></i>
                        </button>
                    </div>
                </div>
                
                <div class="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 2xl:grid-cols-5 gap-6 pb-4">
            `;

            pageItems.forEach(company => {
                html += `
                    <div onclick="openModal('${company.id}')" class="glass-card p-6 rounded-xl cursor-pointer hover:bg-white/5 transition-all hover:-translate-y-1 group border-t-2 border-transparent hover:border-${company.color.split('-')[1]}-500">
                        <div class="flex justify-between items-start mb-4">
                            <div class="w-12 h-12 rounded-lg bg-dark-900 flex items-center justify-center border border-white/5 group-hover:border-${company.color.split('-')[1]}-500/50 transition-colors">
                                <i class="fa-solid ${company.icon} text-xl ${company.color}"></i>
                            </div>
                            <i class="fa-solid fa-arrow-up-right-from-square text-slate-600 text-xs group-hover:text-white transition-colors"></i>
                        </div>
                        <h4 class="text-white font-bold mb-1 truncate">${company.name}</h4>
                        <p class="text-xs text-slate-400 uppercase tracking-wider mb-3">${company.specialty}</p>
                        <p class="text-sm text-slate-500 line-clamp-2">${company.description}</p>
                    </div>
                `;
            });

            html += `</div>`;
            container.innerHTML = html;
        }

        function changePage(category, direction) {
            const companies = allCompanies.filter(c => c.category === category);
            const totalPages = Math.ceil(companies.length / state[category].itemsPerPage);
            
            const newPage = state[category].page + direction;
            if (newPage >= 1 && newPage <= totalPages) {
                state[category].page = newPage;
                renderCategory(category);
            }
        }

        // --- MODAL LOGIC ---
        function openModal(companyId) {
            const company = allCompanies.find(c => c.id === companyId);
            if (!company) return;

            document.getElementById('modal-title').innerText = company.name;
            document.getElementById('modal-category').innerText = company.category;
            document.getElementById('modal-desc').innerText = company.description + " This partner offers comprehensive support for enterprise integration, ensuring scalability and security across your AI infrastructure.";
            document.getElementById('modal-contact').innerText = company.contact;
            document.getElementById('modal-specialty').innerText = company.specialty;
            document.getElementById('modal-link').href = company.website;
            
            const iconContainer = document.getElementById('modal-icon-container');
            iconContainer.innerHTML = `<i class="fa-solid ${company.icon} text-3xl ${company.color}"></i>`;

            const modal = document.getElementById('detail-modal');
            modal.classList.remove('hidden');
            // Small delay to allow display:block to apply before opacity transition
            setTimeout(() => {
                modal.classList.add('visible');
            }, 10);
        }

        function closeModal() {
            const modal = document.getElementById('detail-modal');
            modal.classList.remove('visible');
            setTimeout(() => {
                modal.classList.add('hidden');
            }, 300);
        }

        // --- CHART LOGIC ---
        function initHomeChart() {
            const ctx = document.getElementById('homeChart').getContext('2d');
            
            // Gradient
            const gradient = ctx.createLinearGradient(0, 0, 0, 400);
            gradient.addColorStop(0, 'rgba(14, 165, 233, 0.5)');
            gradient.addColorStop(1, 'rgba(14, 165, 233, 0)');

            window.homeChartInstance = new Chart(ctx, {
                type: 'line',
                data: {
                    labels: ['Week 1', 'Week 2', 'Week 3', 'Week 4', 'Week 5'],
                    datasets: [{
                        label: 'System Reliability (%)',
                        data: [65, 78, 85, 92, 98],
                        borderColor: '#0ea5e9',
                        backgroundColor: gradient,
                        borderWidth: 3,
                        pointBackgroundColor: '#0f172a',
                        pointBorderColor: '#0ea5e9',
                        pointBorderWidth: 2,
                        pointRadius: 4,
                        fill: true,
                        tension: 0.4
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: {
                            labels: { color: '#94a3b8' }
                        }
                    },
                    scales: {
                        y: {
                            grid: { color: 'rgba(255, 255, 255, 0.05)' },
                            ticks: { color: '#64748b' },
                            beginAtZero: true,
                            max: 100
                        },
                        x: {
                            grid: { display: false },
                            ticks: { color: '#64748b' }
                        }
                    }
                }
            });
        }

        // Close modal on Escape key
        document.addEventListener('keydown', function(event) {
            if (event.key === "Escape") {
                closeModal();
            }
        });

        // Initialize
        document.addEventListener('DOMContentLoaded', () => {
            showView('home');
        });

    </script>
</body>
