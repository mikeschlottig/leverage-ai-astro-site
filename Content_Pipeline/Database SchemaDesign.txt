        const apiKey = ""; // Provided by execution environment

        // Gemini API Helper with Exponential Backoff
        async function callGemini(prompt, systemInstruction = "") {
            const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
            const payload = {
                contents: [{ parts: [{ text: prompt }] }],
                systemInstruction: { parts: [{ text: systemInstruction }] }
            };

            const maxRetries = 5;
            for (let i = 0; i < maxRetries; i++) {
                try {
                    const response = await fetch(url, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });
                    
                    if (!response.ok) throw new Error(`HTTP Error: ${response.status}`);
                    
                    const data = await response.json();
                    return data.candidates?.[0]?.content?.parts?.[0]?.text || "No response generated.";
                } catch (error) {
                    if (i === maxRetries - 1) throw error;
                    const delay = Math.pow(2, i) * 1000;
                    await new Promise(resolve => setTimeout(resolve, delay));
                }
            }
        }

            btn.disabled = true;
            btn.innerHTML = '<span class="loader"></span> <span>Analyzing...</span>';
            outputDiv.classList.add('hidden');

            try {
                const systemPrompt = "You are a database architect. Based on the project description, recommend one of three paradigms: Relational (SQL), Node-based (Graph), or Object-based (NoSQL). Briefly justify why based on the trade-offs of performance, flexibility, and complexity. Keep it concise (max 150 words).";
                const response = await callGemini(input, systemPrompt);
                
                outputDiv.innerHTML = response.replace(/\n/g, '<br>');
                outputDiv.classList.remove('hidden');
            } catch (error) {
                outputDiv.innerHTML = `<span class="text-red-500 italic">Error connecting to ✨ AI Architect. Please try again.</span>`;
                outputDiv.classList.remove('hidden');
            } finally {
                btn.disabled = false;

        }

                options: {
                    maintainAspectRatio: false,
                    responsive: true,
                    scales: {
                        y: {

                        }

                    plugins: { legend: { position: 'top' } }
                }
            });

        window.setAppView = function(appId) {
            ['twenty', 'attio', 'superagi'].forEach(id => {
                const btn = document.getElementById(`btn-${id}`);
                btn.classList.toggle('bg-white', id === appId);
                btn.classList.toggle('shadow-sm', id === appId);
                btn.classList.toggle('text-slate-800', id === appId);
                btn.classList.toggle('text-slate-500', id !== appId);
            });

            const data = appData[appId];
            let colors = { twenty: 'text-sky-600', attio: 'text-emerald-600', superagi: 'text-violet-600' };
            
            document.getElementById('app-content').innerHTML = `
                <h3 class="text-2xl font-bold text-slate-900 mb-2">${data.title}</h3>
                <p class="${colors[appId]} font-bold mb-4 uppercase tracking-wide text-xs">${data.subtitle}</p>
                <p class="text-slate-600 mb-4 text-sm leading-relaxed">${data.desc}</p>
                <div class="space-y-3">
                    ${data.bullets.map(b => `
                        <div class="bg-white p-3 rounded border border-slate-200 shadow-sm">
                            <strong class="text-sm text-slate-800 block mb-1">${b.title}</strong>
                            <span class="text-xs text-slate-500 leading-snug block">${b.text}</span>
                        </div>
                    `).join('')}
                </div>
            `;

            let chartColors = { twenty: '14, 165, 233', attio: '16, 185, 129', superagi: '139, 92, 246' };
            appRadarChartInst.data.datasets[0].data = data.radar;
            appRadarChartInst.data.datasets[0].backgroundColor = `rgba(${chartColors[appId]}, 0.2)`;
            appRadarChartInst.data.datasets[0].borderColor = `rgba(${chartColors[appId]}, 1)`;
            appRadarChartInst.data.datasets[0].pointBackgroundColor = `rgba(${chartColors[appId]}, 1)`;
            appRadarChartInst.update();
        };
