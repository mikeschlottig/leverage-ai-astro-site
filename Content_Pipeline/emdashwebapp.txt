class AutoSEOPlugin extends EmDashPlugin {
    constructor(app) {
        super(app);
        this.name = 'AutoSEO';
    }

    async initialize() {
        // Hook into the document lifecycle
        this.app.hooks.on('document.preSave', async (doc) => {
            if (doc.type === 'article' && !doc.seoDescription) {
                // Generate simple snippet
                doc.seoDescription = doc.content.substring(0, 150) + '...';
                this.app.logger.info(`Generated SEO for ${doc.id}`);
            }
            return doc;
        });
    }
}

export default AutoSEOPlugin;</code></pre>

                        <pre class="text-emerald-400 font-mono text-sm overflow-x-auto" id="pipelineCode"><code>import asyncio
import trafilatura
from crawl4ai import AsyncWebCrawler
import requests

EMDASH_API = "https://api.emdash.io/v1/content"
HEADERS = {"Authorization": "Bearer YOUR_API_KEY"}

async def extract_and_push(url):
    # 1. Crawl complex site async
    crawler = AsyncWebCrawler()
    await crawler.start()
    result = await crawler.arun(url=url)
    
    # 2. Extract clean text/metadata using Trafilatura
    downloaded = trafilatura.fetch_url(url)
    content_dict = trafilatura.extract(
        downloaded, 
        output_format="json", 
        with_metadata=True
    )
    
    # 3. Push to EmDash CMS via REST
    if content_dict:
        payload = {
            "title": content_dict.get("title", "Untitled"),
            "body": content_dict.get("text", ""),
            "author": content_dict.get("author", "System"),
            "status": "draft"
        }
        
        response = requests.post(EMDASH_API, json=payload, headers=HEADERS)
        print(f"EmDash Ingest: {response.status_code} - {url}")

    await crawler.close()

# Execute Pipeline
asyncio.run(extract_and_push("https://example.com/target-article"))</code></pre>

            const codeBlock = document.getElementById(elementId);
            navigator.clipboard.writeText(codeBlock.innerText).then(() => {
                const btn = codeBlock.previousElementSibling;
                const originalText = btn.innerText;
                btn.innerText = 'Copied!';
                btn.classList.add('bg-emerald-600');
                setTimeout(() => {
                    btn.innerText = originalText;
                    btn.classList.remove('bg-emerald-600');
                }, 2000);
            }).catch(err => {
                console.error('Failed to copy: ', err);
            });
        }
    </script>
