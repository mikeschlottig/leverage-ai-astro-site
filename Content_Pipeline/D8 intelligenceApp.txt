// Gemini API Integration Helpers
const apiKey = ""; // Provided by environment
const GEMINI_MODEL = "gemini-2.5-flash-preview-09-2025";

const callGemini = async (prompt, systemInstruction = "") => {
  const url = `https://generativelanguage.googleapis.com/v1beta/models/${GEMINI_MODEL}:generateContent?key=${apiKey}`;
  
  const payload = {
    contents: [{ parts: [{ text: prompt }] }],
    systemInstruction: { parts: [{ text: systemInstruction }] }
  };

  const fetchWithRetry = async (retries = 0) => {
    try {
      const response = await fetch(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });
      if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
      return await response.json();
    } catch (err) {
      if (retries < 5) {
        const delay = Math.pow(2, retries) * 1000;
        await new Promise(resolve => setTimeout(resolve, delay));
        return fetchWithRetry(retries + 1);
      }
      throw err;
    }
  };

  const result = await fetchWithRetry();
  return result.candidates?.[0]?.content?.parts?.[0]?.text || "No response generated.";
};

                </div>
              ))}
            </div>
          </div>
        )}

      </main>

      {/* ✨ AI Research Assistant Panel */}
      <div className={`fixed bottom-8 right-8 z-[60] transition-all duration-500 transform ${isAiOpen ? 'translate-y-0 opacity-100 scale-100' : 'translate-y-12 opacity-0 scale-95 pointer-events-none'}`}>
        <div className={`w-[400px] h-[600px] rounded-[32px] shadow-2xl overflow-hidden border flex flex-col ${darkMode ? 'bg-slate-900 border-slate-800 text-white' : 'bg-white border-slate-200 text-slate-900'}`}>
          <div className="p-6 bg-indigo-600 text-white flex items-center justify-between">
            <div className="flex items-center gap-2">
              <Sparkles size={20} />
              <h3 className="font-bold">✨ AI Research Assistant</h3>
            </div>
            <button onClick={() => setIsAiOpen(false)}><X size={20} /></button>
          </div>
          
          <div className="flex-1 p-6 overflow-y-auto space-y-4">
            {!aiResponse && !isAiLoading && (
              <div className="text-center py-10">
                <div className="bg-indigo-500/10 w-16 h-16 rounded-full flex items-center justify-center mx-auto mb-4">
                  <MessageSquare className="text-indigo-500" />
                </div>
                <h4 className="font-bold mb-2">How can I help you today?</h4>
                <p className="text-sm text-slate-400 px-4">Ask about market trends, vendor comparisons, or the 2026 Farm Bill impact.</p>
                <div className="flex flex-wrap gap-2 justify-center mt-6">
                  {["Summarize the Farm Bill", "Compare Top 3 Vendors", "2026 Growth Trend"].map(hint => (
                    <button key={hint} onClick={() => {setAiQuery(hint); handleAiSearch();}} className={`text-xs px-3 py-1.5 rounded-full border ${darkMode ? 'border-slate-800 hover:bg-slate-800' : 'border-slate-200 hover:bg-slate-100'}`}>
                      {hint}
                    </button>
                  ))}
                </div>
              </div>
            )}
            
            {isAiLoading && (
              <div className="flex flex-col items-center justify-center py-10 gap-3">
                <Loader2 className="text-indigo-500 animate-spin" size={32} />
                <p className="text-sm font-medium animate-pulse">✨ Analyzing research data...</p>
              </div>
            )}
            
            {aiResponse && (
              <div className="animate-in fade-in slide-in-from-bottom-2 duration-300">
                <div className="text-xs font-bold text-indigo-400 mb-2 uppercase tracking-widest flex items-center gap-2">
                  <Sparkles size={12} /> ✨ AI Research Synthesis
                </div>
                <div className={`p-5 rounded-2xl text-sm leading-relaxed border ${darkMode ? 'bg-slate-950/50 border-indigo-500/20 text-slate-300' : 'bg-indigo-50 border-indigo-100 text-slate-700'}`}>
                  {aiResponse}
                </div>
              </div>
            )}
            
            {aiError && (
              <div className="p-4 rounded-xl bg-red-500/10 border border-red-500/20 text-red-500 text-sm flex items-start gap-2">
                <AlertCircle size={16} className="shrink-0 mt-0.5" />
                {aiError}
              </div>
            )}
          </div>
          
          <div className={`p-6 border-t ${darkMode ? 'bg-slate-950 border-slate-800' : 'bg-slate-50 border-slate-200'}`}>
            <div className="relative">
              <input 
                type="text" 
                value={aiQuery}
                onChange={(e) => setAiQuery(e.target.value)}
                onKeyDown={(e) => e.key === 'Enter' && handleAiSearch()}
                placeholder="Ask intelligence assistant..."
                className={`w-full p-4 pr-12 rounded-2xl border text-sm transition-all focus:ring-2 focus:ring-indigo-500 outline-none ${darkMode ? 'bg-slate-900 border-slate-800 text-white' : 'bg-white border-slate-300'}`}
              />
              <button 
                onClick={handleAiSearch}
                disabled={isAiLoading || !aiQuery}
                className="absolute right-3 top-1/2 -translate-y-1/2 p-2 bg-indigo-600 text-white rounded-xl disabled:opacity-50"
              >
                <Send size={16} />
              </button>
            </div>
          </div>
        </div>
      </div>

      {/* Floating Trigger Button */}
      {!isAiOpen && (
        <button 
          onClick={() => setIsAiOpen(true)}
          className="fixed bottom-8 right-8 z-[60] w-16 h-16 bg-indigo-600 text-white rounded-full shadow-2xl flex items-center justify-center hover:scale-110 active:scale-95 transition-all group"
        >
          <Sparkles className="group-hover:rotate-12 transition-transform" />
          <div className="absolute -top-1 -right-1 w-4 h-4 bg-emerald-500 rounded-full border-2 border-slate-950 animate-pulse"></div>
        </button>
      )}

      {/* Modals */}
      <Modal item={selectedItem} onClose={() => setSelectedItem(null)} type="vendor" />
      <Modal item={selectedSource} onClose={() => setSelectedSource(null)} type="source" />
    </div>
  );
};

export default App;
