#### PocketFlow Framework
```typescript
// Core Classes
class Node<SharedStorage> {
  async prep(shared: S): Promise<unknown>    // Prepare data
  async exec(prepRes: unknown): Promise<unknown>  // Execute logic
  async post(shared: S, prepRes, execRes): Promise<string | undefined>  // Finalize & route
  
  // Action-based routing
  next(node: Node)  // Default transition
  on(action: string, node: Node)  // Conditional transition
}

// Flow orchestration
class Flow extends Node {
  async run(shared: S)  // Execute the entire flow
}
```

## Recommended Implementation: API Key & Model Selection

### Database Schema Extension

```prisma
// Add to prisma/schema.prisma

model APIKey {
  id        String   @id @default(cuid())
  provider  String   // "openai", "anthropic", "zai", etc.
  name      String   // User-friendly name
  key       String   // Encrypted API key
  isActive  Boolean  @default(true)
  userId    String?  // For future auth
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
}

model ModelConfig {
  id           String   @id @default(cuid())
  provider     String
  model        String   // "gpt-4", "claude-3-opus", etc.
  displayName  String
  capabilities String   // JSON: { vision: true, tools: true }
  isDefault    Boolean  @default(false)
  isActive     Boolean  @default(true)
}

model AgentConfig {
  id           String   @id @default(cuid())
  agentId      String   @unique
  agent        Agent    @relation(fields: [agentId], references: [id])
  apiKeyId     String?
  apiKey       APIKey?  @relation(fields: [apiKeyId], references: [id])
  modelId      String?
  model        ModelConfig? @relation(fields: [modelId], references: [id])
  temperature  Float    @default(0.7)
  maxTokens    Int      @default(4096)
  systemPrompt String?
}
```
