    <script>
        // Register ScrollTrigger
        gsap.registerPlugin(ScrollTrigger);

        // Progress Bar
        window.addEventListener('scroll', () => {
            const winScroll = document.body.scrollTop || document.documentElement.scrollTop;
            const height = document.documentElement.scrollHeight - document.documentElement.clientHeight;
            const scrolled = (winScroll / height) * 100;
            document.getElementById('progressBar').style.width = scrolled + "%";
        });

        // Navigation Active State
        const sections = document.querySelectorAll('section[id]');
        const navLinks = document.querySelectorAll('.nav-link');

        window.addEventListener('scroll', () => {
            let current = '';
            sections.forEach(section => {
                const sectionTop = section.offsetTop;
                const sectionHeight = section.clientHeight;
                if (pageYOffset >= (sectionTop - 200)) {
                    current = section.getAttribute('id');
                }
            });

            navLinks.forEach(link => {
                link.classList.remove('active');
                if (link.getAttribute('href').slice(1) === current) {
                    link.classList.add('active');
                }
            });
        });

        // View Switching Logic
        function switchView(viewName) {
            // Hide all views
            ['lowcode', 'opensource', 'framework'].forEach(v => {
                document.getElementById(`view-${v}`).classList.add('hidden');
                document.getElementById(`btn-${v}`).classList.remove('active');
                document.getElementById(`btn-${v}`).classList.add('text-gray-400');
            });
            
            // Show selected view
            document.getElementById(`view-${viewName}`).classList.remove('hidden');
            document.getElementById(`btn-${viewName}`).classList.add('active');
            document.getElementById(`btn-${viewName}`).classList.remove('text-gray-400');
        }

        // GSAP Animations
        // Hero text reveal
        gsap.from("header h1", {
            duration: 1.2,
            y: 100,
            opacity: 0,
            ease: "power4.out",
            delay: 0.2
        });

        gsap.from("header p", {
            duration: 1,
            y: 30,
            opacity: 0,
            ease: "power3.out",
            delay: 0.6
        });

        // Section reveals
        gsap.utils.toArray('section').forEach(section => {
            gsap.from(section.querySelectorAll('h2, h3, p, .insight-card, table'), {
                scrollTrigger: {
                    trigger: section,
                    start: "top 80%",
                    toggleActions: "play none none reverse"
                },
                y: 30,
                opacity: 0,
                duration: 0.8,
                stagger: 0.1,
                ease: "power3.out"
            });
        });

        // Parallax effect for decorative elements
        gsap.to(".grid-bg", {
            scrollTrigger: {
                trigger: "body",
                start: "top top",
                end: "bottom bottom",
                scrub: 1
            },
            backgroundPosition: "0 100%"
        });

        // Smooth scroll for anchor links
        document.querySelectorAll('a[href^="#"]').forEach(anchor => {
            anchor.addEventListener('click', function (e) {
                e.preventDefault();
                const target = document.querySelector(this.getAttribute('href'));
                if (target) {
                    target.scrollIntoView({
                        behavior: 'smooth',
                        block: 'start'
                    });
                }
            });
        });
    </script>
