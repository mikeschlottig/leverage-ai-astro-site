        function nav(sectionId, btn) {
            document.querySelectorAll('.content-section').forEach(el => el.classList.add('hidden'));
            document.getElementById(sectionId).classList.remove('hidden');
            
            document.querySelectorAll('.nav-btn').forEach(el => {
                el.classList.remove('nav-active');
            });
            btn.classList.add('nav-active');

            if(sectionId === 'economics' && !window.chartRendered) {
                renderChart();
                window.chartRendered = true;
            }
        }

        function switchTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.getElementById(tabId).classList.remove('hidden');

            document.querySelectorAll('.tab-btn').forEach(el => el.classList.remove('tab-active'));
            document.getElementById('btn-' + tabId).classList.add('tab-active');
        }

        function toggleAccordion(id, btn) {
            const content = document.getElementById(id);
            const icon = btn.querySelector('.acc-icon');
            if (content.classList.contains('hidden')) {
                content.classList.remove('hidden');
                icon.textContent = '➖';
            } else {
                content.classList.add('hidden');
                icon.textContent = '➕';
            }
        }

        function copyCode(element) {
            const code = element.nextElementSibling.innerText;
            navigator.clipboard.writeText(code).then(() => {
                const originalText = element.innerText;
                element.innerText = '✅';
                setTimeout(() => { element.innerText = originalText; }, 2000);
            });
        }

        // --- Gemini API Integrations ---
        const apiKey = ""; 

        async function callGemini(prompt) {
            const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
            const payload = {
                contents: [{ parts: [{ text: prompt }] }]
            };
            
            let retries = 5;
            let delay = 1000;
            while (retries > 0) {
                try {
                    const response = await fetch(url, {
                        method: "POST",
                        headers: { "Content-Type": "application/json" },
                        body: JSON.stringify(payload)
                    });
                    if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
                    const result = await response.json();
                    return result.candidates?.[0]?.content?.parts?.[0]?.text || "No response generated.";
                } catch (error) {
                    retries--;
                    if (retries === 0) return "Error generating response. Please try again.";
                    await new Promise(r => setTimeout(r, delay));
                    delay *= 2;
                }
            }
        }

            btn.disabled = true;

Keep it professional and punchy. No markdown formatting, just plain text.";

        }

        async function generateCode() {
            const btn = document.getElementById('btn-generate-code');
            const input = document.getElementById('ai-prompt').value;
            const outputContainer = document.getElementById('code-output-container');
            const output = document.getElementById('code-output');
            
            if(!input.trim()) {
                outputContainer.classList.remove('hidden');
                output.innerText = "Please enter a task description first.";
                return;
            }

            btn.disabled = true;
            btn.innerHTML = '✨ Generating...';
            outputContainer.classList.remove('hidden');
            output.innerText = 'Writing script...';

            const prompt = `You are a Python developer. Write a complete Python script using the 'trafilatura' library based on the following request: "${input}". Only output the code, without Markdown backticks surrounding the whole block. Include brief comments.`;
            
            let response = await callGemini(prompt);
            
            // Clean up any stray markdown code fences if the model happens to return them
            response = response.replace(/^```python\n/gi, '').replace(/^```\n/gi, '').replace(/```$/gi, '');
            
            output.innerText = response;
            
            btn.disabled = false;
            btn.innerHTML = '✨ Generate Code';
        }
